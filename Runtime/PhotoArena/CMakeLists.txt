cmake_minimum_required(VERSION 3.20)
project(WaifuPhotoArena LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET 15.0)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PHOTOARENA_LLAMA_SOURCE}" llama)
add_executable(waifu-photoarena main.mm)
target_link_libraries(waifu-photoarena PRIVATE mtmd llama "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics")
target_compile_options(waifu-photoarena PRIVATE -fobjc-arc)

# Opt-in correctness executable: callers place it in temporary storage.
if(PHOTOARENA_TEST_OUTPUT)
    add_executable(photoarena-fusion-tests EXCLUDE_FROM_ALL tests/fusion.cpp)
    target_link_libraries(photoarena-fusion-tests PRIVATE ggml ggml-metal)
    set_target_properties(photoarena-fusion-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PHOTOARENA_TEST_OUTPUT}")
    add_executable(photoarena-order-tests EXCLUDE_FROM_ALL tests/order.mm)
    target_link_libraries(photoarena-order-tests PRIVATE mtmd llama "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics")
    target_compile_options(photoarena-order-tests PRIVATE -fobjc-arc)
    set_target_properties(photoarena-order-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PHOTOARENA_TEST_OUTPUT}")
    add_executable(photoarena-sampling-tests EXCLUDE_FROM_ALL tests/sampling.mm)
    target_link_libraries(photoarena-sampling-tests PRIVATE mtmd llama "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics")
    target_compile_options(photoarena-sampling-tests PRIVATE -fobjc-arc)
    set_target_properties(photoarena-sampling-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PHOTOARENA_TEST_OUTPUT}")
endif()
